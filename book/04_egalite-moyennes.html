### Régions d'acceptation et de rejet de l'hypothèse nulle

```{r}
#| label: student_distrib_fig
#| fig-cap: "**Régions d'acceptation et de rejet de l'hypothèse nulle dans le cadre du test de Student**, illustrées par une distribution avec 10 degrés de liberté. Panneaux de gauche : fonction de densité. Panneaux de droite : function de répartition (Cumularive Distribution Function, CDF). Panneaux du haut : test bilatéral ; panneaux du milieu : test unilatéral à droite ; panneaux du bas : test unilatéral à gauche. Flèches vertes : régions d'acceptation ; flèches rouges : régions de rejet. "
#| echo: false
#| warning: false
#| message: false
#| fig-width: 8
#| fig-height: 7
#| out-width: "100%"

## Panel configuration
par(mfrow=c(3,2))


## Parameters
alpha <- 0.05
df <- 10
t_threshold <- abs(qt(p = alpha, df = df))
x <- seq(from=-4, to=4, by=0.01)
dens <- dt(x, df = df) ## Density
cdf <- pt(x, df = 10, lower.tail = TRUE) ## Cumulative density function

## Two-tailed test, density
plot(x, 
     dens, 
     main = paste0("Density, bilateral test"),
     xlab = "t",
     ylab = "d(t)",
     ylim = c(0, max(dens)*1.3),
     type='l', 
     col = "blue",
     las = 1)
grid()
x_up = seq(from=t_threshold, to = 4.0, by=0.01)
x_mid = seq(from=-t_threshold, to = t_threshold, by=0.01)
x_down = seq(from=-4.0, to = -t_threshold, by=0.01)
lines(x_up, dt(x_up, df = df), type = "h", col = "#FFBBBB")
lines(x_mid, dt(x_mid, df = df), type = "h", col = "#BBFFBB")
lines(x_down, dt(x_down, df = df), type = "h", col = "#FFBBBB")
abline(v=0, col="#BBBBBB")
abline(v = c(-t_threshold, t_threshold),   col = "red")
arrows(x0=-4, x1 = -t_threshold, y0 = max(dens)*1.2, y1 = max(dens)*1.2, col = "red", length = 0.1, code = 2, lwd = 2)
arrows(x0=t_threshold, x1 = 4, y0 = max(dens)*1.2, y1 = max(dens)*1.2, col = "red", length = 0.1, code = 1, lwd=2)
arrows(x0=-t_threshold, x1 = t_threshold, y0 = max(dens)*1.2, y1 = max(dens)*1.2,  col = "#00BB00", length = 0.1, code = 3, lwd=2)


## Two-tailed test, CDF
plot(x, 
     cdf, 
     main = "CDF, bilateral test",
     xlab = "t",
     ylab = expression(P(T <= t)),
     ylim = c(0,1.3),
     type='l', 
     col = "darkviolet",
     las = 1)
grid()
abline(v=0, col="#BBBBBB")
abline(v = c(-t_threshold, t_threshold),  col = "red")
arrows(x0=-4, x1 = -t_threshold, y0 = 1.2, y1 = 1.2, col = "red", length = 0.1, code = 2, lwd = 2)
arrows(x0=t_threshold, x1 = 4, y0 = 1.2, y1 = 1.2, col = "red", length = 0.1, code = 1, lwd=2)
arrows(x0=-t_threshold, x1 = t_threshold, y0 = 1.2, y1 = 1.2, col = "#00BB00", length = 0.1, code = 3, lwd=2)


## Right-sided test, density
plot(x, 
     dens, 
     main = paste0("Density, right-sided test"),
     xlab = "t",
     ylab = "d(t)",
     ylim = c(0, max(dens)*1.3),
     type='l', 
     col = "blue",
     las = 1)
grid()
x_reject = seq(from=t_threshold, to = 4.0, by=0.01)
x_accept = seq(from=-4, to = t_threshold, by=0.01)
lines(x_reject, dt(x_reject, df = df), type = "h", col = "#FFBBBB")
lines(x_accept, dt(x_accept, df = df), type = "h", col = "#BBFFBB")
abline(v=0, col="#BBBBBB")
abline(v = t_threshold,   col = "red")
arrows(x0=-4, x1 = t_threshold, y0 = max(dens)*1.2, y1 = max(dens)*1.2, col = "#00BB00", length = 0.1, code = 2, lwd=2)
arrows(x0=t_threshold, x1 = 4, y0 = max(dens)*1.2, y1 = max(dens)*1.2, col = "red", length = 0.1, code = 1, lwd=2)


## Right-sided test, CDF
plot(x, 
     cdf, 
     main = "CDF, right-sided test",
     xlab = "t",
     ylab = expression(P(T <= t)),
     ylim = c(0,1.3),
     type='l', 
     col = "darkviolet",
     las = 1)
grid()
abline(v=0, col="#BBBBBB")
abline(v = t_threshold,  col = "red")
arrows(x0=-4, x1 = t_threshold, y0 = 1.2, y1 = 1.2, col = "#00BB00", length = 0.1, code = 2, lwd=2)
arrows(x0=t_threshold, x1 = 4, y0 = 1.2, y1 = 1.2, col = "red", length = 0.1, code = 1, lwd=2)


## Left-sided test, density
plot(x, 
     dens, 
     main = paste0("Density, left-sided test"),
     xlab = "t",
     ylab = "d(t)",
     ylim = c(0, max(dens)*1.3),
     type='l', 
     col = "blue",
     las = 1)
grid()
x_reject = seq(from = -4, to = -t_threshold, by = 0.01)
x_accept = seq(from = -t_threshold, to = 4, by = 0.01)
lines(x_reject, dt(x_reject, df = df), type = "h", col = "#FFBBBB")
lines(x_accept, dt(x_accept, df = df), type = "h", col = "#BBFFBB")
abline(v=0, col="#BBBBBB")
abline(v = -t_threshold,   col = "red")
arrows(x0=-4, x1 = -t_threshold, y0 = max(dens)*1.2, y1 = max(dens)*1.2, col = "red", length = 0.1, code = 2, lwd = 2)
arrows(x0=-t_threshold, x1 =4, y0 = max(dens)*1.2, y1 = max(dens)*1.2, col = "#00BB00", length = 0.1, code = 1, lwd=2)

## Left-tailed test, CDF
plot(x, 
     cdf, 
     main = "CDF, left-sided test",
     xlab = "t",
     ylab = expression(P(T <= t)),
     ylim = c(0,1.3),
     type='l', 
     col = "darkviolet",
     las = 1)
grid()
abline(v=0, col="#BBBBBB")
abline(v = -t_threshold,  col = "red")
arrows(x0=-4, x1 = -t_threshold, y0 = 1.2, y1 = 1.2, col = "red", length = 0.1, code = 2, lwd = 2)
arrows(x0=-t_threshold, x1 =4, y0 = 1.2, y1 = 1.2, col = "#00BB00", length = 0.1, code = 1, lwd=2)
```
