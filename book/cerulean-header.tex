\usepackage{xcolor}
\usepackage{sectsty}
\usepackage{titlesec}

% Cerulean colors
\definecolor{cerulean}{HTML}{2FA4E7}
\definecolor{ceruleandark}{HTML}{157AB5}

% Section headings
\sectionfont{\color{cerulean}}
\subsectionfont{\color{ceruleandark}}

  {\large\bfseries\color{ceruleandark}}

  {1em}

\fancyhf{}

}

% Configure hyperlinks after hyperref has been loaded

  \hypersetup{

    linkcolor=ceruleandark,
