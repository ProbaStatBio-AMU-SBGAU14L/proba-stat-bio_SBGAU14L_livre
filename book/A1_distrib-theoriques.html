### Tableau de valeurs critiques bilatérales de la distribution de Student

```{r}
#| label: tbl-student-bilateral
#| tbl-cap: '**Valeurs critiques bilatérales de la distribution de Student.** La première colonne indique les degrés de liberté (d.d.l.). La table fournit les valeurs de $t$ telles que $P(\lvert T_\\nu \rvert \geq t) = \alpha$. '
#| echo: false
#| warning: false
#| message: false

library(knitr)
# library(kableExtra)

# Probabilités bilatérales indiquées en tête de colonne
alpha <- c(0.90, 0.50, 0.30, 0.20, 0.10,
           0.05, 0.02, 0.01, 0.001)

# Degrés de liberté
ddl <- 1:30

# Valeurs critiques : P(|T| >= t) = alpha
student_values <- outer(
  ddl,
  alpha,
  FUN = function(nu, a) qt(1 - a / 2, df = nu)
)

# Limite pour un nombre infini de degrés de liberté :
# distribution normale centrée réduite
normal_values <- qnorm(1 - alpha / 2)

student_table <- rbind(
  student_values,
  normal_values
)

student_table <- data.frame(
  "d.d.l." = c(as.character(ddl), "\u221e"),
  student_table,
  check.names = FALSE
)

colnames(student_table)[-1] <- c(
  sprintf("%.2f", alpha[-length(alpha)]),
  sprintf("%.3f", alpha[length(alpha)])
)


## Detect the current output format
is_latex_output <- knitr::is_latex_output()
is_html_output <- knitr::is_html_output()

## Select a table format compatible with the target document
table_format <- if (is_latex_output) {
  "latex"
} else if (is_html_output) {
  "html"
} else {
  ## Use a Pandoc Markdown table for DOCX and other non-HTML formats
  "pipe"
}

## Build the table in the format appropriate for the current output
student_table_output <- knitr::kable(
  student_table,
  format = table_format,
  align = c("r", rep("r", length(alpha))),
  booktabs = is_latex_output,
  longtable = FALSE,
  row.names = FALSE,
  escape = FALSE,
  format.args = list(
    decimal.mark = ".",
    big.mark = "\u202F",
    scientific = FALSE
  )
)

## Apply kableExtra styling only to formats supported by kableExtra
if (is_html_output || is_latex_output) {
  student_table_output <- kableExtra::kable_styling(
    student_table_output,
    full_width = FALSE
  )
}

## Print the resulting table
student_table_output
```
